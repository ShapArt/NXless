cmake_minimum_required(VERSION 3.31)
project(nxless LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NXLESS_BUILD_TESTS "Build host tests" ON)
option(NXLESS_ENABLE_SANITIZERS "Enable ASan/UBSan on supported host compilers" ON)
add_subdirectory(common)
if(NXLESS_BUILD_TESTS)
  include(CTest)
  enable_testing()
  add_subdirectory(tests)
endif()
